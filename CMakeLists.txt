cmake_minimum_required(VERSION 3.16)
project(la_mapf_solvers LANGUAGES CXX)
add_executable(la_solver solvers/main.cpp)
target_compile_features(la_solver PRIVATE cxx_std_17)
set_target_properties(la_solver PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(la_solver PRIVATE /W4 /utf-8)
else()
  target_compile_options(la_solver PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter QUIET)
if(Python3_Interpreter_FOUND)
  add_test(NAME solver_regression COMMAND "${Python3_EXECUTABLE}" "${CMAKE_SOURCE_DIR}/tests/test_solvers.py" "$<TARGET_FILE:la_solver>")
endif()
